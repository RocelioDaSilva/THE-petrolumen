\section*{Plano de Ação Sugerido (Começar pelos Mais Fáceis)}
\addcontentsline{toc}{section}{Plano de Ação Sugerido (Começar pelos Mais Fáceis)}

\subsection*{Semana 1--2: Sensibilização (Nível 1)}
\begin{enumerate}[leftmargin=*]
    \item \textbf{PPE / First Aid / Defensive Driving} (SafetyCulture) --- 0,5h cada
    \item \textbf{Introduction to Drilling} (LearnToDrill) --- 1h
    \item \textbf{H2S Awareness} (Alison) --- 2,25h
    \item \textbf{Oil and Gas Production Technology} (Alison) --- 3,5h
\end{enumerate}

\subsection*{Semana 3--4: Fundamentos (Nível 2)}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Level 1 Well Control Awareness} (IWCF) --- 8h (essencial para CV)
    \item \textbf{Petroleum Engineering Principles} (Alison) --- 8h
    \item \textbf{Python for Everybody (PY4E)} --- iniciar (10h+)
\end{enumerate}

\subsection*{Mês 2--3: Intermédio (Nível 3)}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Production Engineering} (Cambridge) --- 12h
    \item \textbf{Python for Oil and Gas} (Mind Luster) --- variável
    \item \textbf{BUOG SPE Fundamentals of Reservoir Engineering} --- 8,5h
\end{enumerate}

\subsection*{Mês 4--6: Avançado (Nível 4)}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Applied Reservoir Engineering} (NPTEL) --- 12 semanas (espinha dorsal)
    \item \textbf{CMG Introduction to Modelling Workflows} --- 2 dias (simulação)
    \item \textbf{SPE Python for Oil \& Gas} --- 22 capítulos
\end{enumerate}

\subsection*{Mês 6--12: Especialização (Nível 4/5)}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Introduction to Python and Petroleum Data Analytics} (NPTEL) --- 12 semanas
    \item \textbf{Python for Reservoir Engineering (GitHub)} --- projetos práticos
    \item \textbf{Earth's Physical Resources: Fossil Fuels} (Open University) --- se quiseres profundidade geológica
\end{enumerate}

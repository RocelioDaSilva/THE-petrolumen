This document contains the complete, runnable source code of PPES.
Every file needed to install, test, and run the software is here.

the software will install with \code{pip install -e .}, pass its tests
with \code{pytest -v}, and run with \code{streamlit run app/streamlit\_app.py}.

"""Appendix A -- unit conversions."""
from enum import Enum

class UnitSystem(str, Enum):
    FIELD = "field"
    SI = "SI"

def api_to_sg(api: float) -> float:
    """Eq. 2.3."""
    return 141.5 / (api + 131.5)

def sg_to_api(gamma_o: float) -> float:
    """Eq. 2.3 inverted."""
    return 141.5 / gamma_o - 131.5

CONVERSIONS = {
    ("ft","m"): 0.3048, ("m","ft"): 3.2808,
    ("in","mm"): 25.4,
    ("bbl","m3"): 0.158987, ("m3","bbl"): 6.2898,
    ("psi","kPa"): 6.89476, ("kPa","psi"): 0.145038,
    ("cp","Pa_s"): 0.001,
    ("md","m2"): 9.86923e-16,
    ("F","C"): lambda f: (f-32)*5/9,
    ("C","F"): lambda c: c*9/5+32,
    ("R","K"): lambda r: r*5/9,
    ("K","R"): lambda k: k*9/5,
}

def convert(value: float, from_unit: str, to_unit: str) -> float:
    key = (from_unit, to_unit)
    if key not in CONVERSIONS:
        raise KeyError(f"No conversion {from_unit}->{to_unit}")
    f = CONVERSIONS[key]
    return f(value) if callable(f) else value * f

"""Chapter 2 -- Properties of Oil and Natural Gas."""
import math
import numpy as np
from scipy.optimize import newton

MW_AIR = 28.97

def rs_standing(p, t_F, api, gamma_g) -> float:
    """Eq. 2.2. [scf/stb]"""
    return gamma_g * ((p/18) * (10**(0.0125*api))
                       / (10**(0.00091*t_F)))**1.2048

def rho_o(p, t_F, api, rs, gamma_g) -> float:
    """Eq. 2.5. [lbm/ft3]"""
    go = 141.5/(api+131.5)
    num = 62.4*go + 0.0136*rs*gamma_g
    den = 0.972 + 0.000147*(rs*math.sqrt(gamma_g/go) + 1.25*t_F)**1.175
    return num/den

def bo_standing(t_F, api, rs, gamma_g) -> float:
    """Eq. 2.7. [rb/stb]"""
    go = 141.5/(api+131.5)
    return 0.9759 + 0.00012*(rs*math.sqrt(gamma_g/go) + 1.25*t_F)**1.2

def mu_dead_oil(api, t_F) -> float:
    """Eqs. 2.8-2.9. [cp]"""
    A = 10**(0.43 + 8.33/api)
    return (0.32 + 1.8e7/api**4.53)*(360.0/(t_F+200.0))**A

def mu_sat_oil(rs, mu_od) -> float:
    """Eqs. 2.10-2.15. [cp]"""
    a = rs*(2.2e-7*rs - 7.4e-4)
    c = 8.62e-5*rs; d = 1.10e-3*rs; e = 3.74e-3*rs
    b = 0.68/(10**c) + 0.25/(10**d) + 0.062/(10**e)
    return 10**a * mu_od**b

def mu_unsat_oil(p, pb, mu_od, mu_ob) -> float:
    """Eq. 2.16. [cp]"""
    return mu_ob + 0.001*(p-pb)*(0.024*mu_od**1.6 + 0.38*mu_ob**0.56)

def oil_compressibility(p, rs, t_F, api, gamma_g) -> float:
    """Eq. 2.17. [1/psi]"""
    c = (-1433 + 5*rs + 17.2*t_F - 1180*gamma_g + 12.61*api) / (1e6*p)
    return max(c, 1e-7)

def z_brill_beggs(p_pr, t_pr) -> float:
    """Eqs. 2.45-2.51."""
    A = 1.39*(t_pr-0.92)**0.5 - 0.36*t_pr - 0.10
    E = 9.0*(t_pr-1.0)
    B = (0.62-0.23*t_pr)*p_pr + (0.066/(t_pr-0.86) - 0.037)*p_pr**2 \
        + 0.32*p_pr**6/10**E
    C = 0.132 - 0.32*math.log10(t_pr)
    F = 0.3106 - 0.49*t_pr + 0.1824*t_pr**2
    D = 10**F
    return A + (1-A)/math.exp(B) + C*p_pr**D

def z_hall_yarborough(p_pr, t_pr) -> float:
    """Eqs. 2.52-2.59."""
    tr = 1.0/t_pr
    A = 0.06125*tr*math.exp(-1.2*(1-tr)**2)
    B = tr*(14.76 - 9.76*tr + 4.58*tr**2)
    C = tr*(90.7 - 242.2*tr + 42.4*tr**2)
    D = 2.18 + 2.82*tr
    def f(Y):
        return (Y + Y**2 + Y**3 - Y**4)/(1-Y)**3 \
               - A*p_pr - B*Y**2 + C*Y**D
    def df(Y):
        return (1+4*Y+4*Y**2-4*Y**3+Y**4)/(1-Y)**4 - 2*B*Y \
               + C*D*Y**(D-1)
    Y = newton(f, x0=0.1, fprime=df, tol=1e-12, maxiter=200)
    return A*p_pr/Y

def pseudo_critical_from_sg(gamma_g):
    """Eqs. 2.22-2.23."""
    return 709.604 - 58.718*gamma_g, 170.491 + 307.344*gamma_g

def pseudo_critical_from_composition(y, p_c, t_c):
    """Eqs. 2.20-2.21."""
    return float(np.sum(y*p_c)), float(np.sum(y*t_c))

def wichert_aziz_correction(y_H2S, y_CO2, p_pc, t_pc):
    """Eqs. 2.24-2.28."""
    A = y_H2S + y_CO2; B = y_H2S
    e3 = 120*(A**0.9 - A**1.6) + 15*(B**0.5 - B**4.0)
    t_new = t_pc - e3
    p_new = p_pc*t_new/(t_pc + B*(1-B)*e3)
    return p_new, t_new

_DEMPSEY = (
    -2.46211820, 2.97054714, -0.28626405, 0.00805420,
     2.80860949, -3.49803305, 0.36037302, -0.01044324,
    -0.79338568, 1.39643306, -0.14914493, 0.00441016,
     0.08393872, -0.18640885, 0.02033679, -0.00060958,
)

def mu_gas_low_pressure(gamma_g, t_R, y_N2=0.0, y_CO2=0.0, y_H2S=0.0):
    """Eqs. 2.36-2.39. [cp]"""
    lg = math.log10(gamma_g)
    mhc = (8.188e-3 - 6.15e-3*lg) + (1.709e-5 - 2.062e-6*gamma_g)*t_R
    return (mhc + (9.59e-3 + 8.48e-3*lg)*y_N2
            + (6.24e-3 + 9.08e-3*lg)*y_CO2
            + (3.73e-3 + 8.49e-3*lg)*y_H2S)

def mu_gas_corr(p_pr, t_pr, mu1):
    """Eqs. 2.40-2.41. [cp]"""
    a = _DEMPSEY
    ln_mr = (a[0]+a[1]*p_pr+a[2]*p_pr**2+a[3]*p_pr**3
             + t_pr*(a[4]+a[5]*p_pr+a[6]*p_pr**2+a[7]*p_pr**3)
             + t_pr**2*(a[8]+a[9]*p_pr+a[10]*p_pr**2+a[11]*p_pr**3)
             + t_pr**3*(a[12]+a[13]*p_pr+a[14]*p_pr**2+a[15]*p_pr**3))
    return (mu1/math.exp(ln_mr))*math.exp(math.exp(ln_mr))

def rho_gas(p, t_R, gamma_g, z):
    """Eq. 2.61. [lbm/ft3]"""
    return 2.7*gamma_g*p/(z*t_R)

def bg(p, t_R, z):
    """Eq. 2.62. [ft3/scf]"""
    return 0.0283*z*t_R/p

def eg(p, t_R, z):
    """Eq. 2.64. [scf/ft3]"""
    return 35.3*p/(z*t_R)

def gas_compressibility(p, z, dz_dp):
    """Eq. 2.68."""
    return 1.0/p - (1.0/z)*dz_dp

"""Chapter 3 -- Reservoir Deliverability / IPR."""
import math
import numpy as np

def j_star_pseudosteady(k, h, bo, mu, A, ca, rw, skin) -> float:
    """Eq. 3.17."""
    return k*h/(141.2*bo*mu*(0.5*math.log(4*A/(1.78*ca*rw**2)) + skin))

def vogel_rate(qmax, pwf, pbar) -> float:
    """Eq. 3.19."""
    r = pwf/pbar
    return qmax*(1 - 0.2*r - 0.8*r*r)

def vogel_pwf(q, qmax, pbar) -> float:
    """Eq. 3.20."""
    return 0.125*pbar*(math.sqrt(max(81-80*q/qmax, 0.0)) - 1)

def qmax_from_j(j_star, pbar) -> float:
    """Eq. 3.21."""
    return j_star*pbar/1.8

def fetkovich_rate(qmax, pwf, pbar, n) -> float:
    """Eq. 3.22."""
    return qmax*(1 - (pwf/pbar)**2)**n

def generalized_vogel(j_star, pbar, pb, pwf) -> float:
    """Eq. 3.28."""
    if pwf >= pb: return j_star*(pbar - pwf)
    qb = j_star*(pbar - pb); qv = j_star*pb/1.8
    r = pwf/pb
    return qb + qv*(1 - 0.2*r - 0.8*r*r)

def transient_ipr_oil(p_i, k, h, bo, mu, phi, ct, rw, skin,
                       t_hours, q) -> float:
    """Eq. 3.1."""
    lt = (math.log10(t_hours) + math.log10(k/(phi*mu*ct*rw**2))
          - 3.23 + 0.87*skin)
    return p_i - 162.6*q*bo*mu*lt/(k*h)

def transient_rate_from_pwf(p_i, k, h, bo, mu, phi, ct, rw, skin,
                             t_hours, pwf) -> float:
    lt = (math.log10(t_hours) + math.log10(k/(phi*mu*ct*rw**2))
          - 3.23 + 0.87*skin)
    return k*h*(p_i - pwf)/(162.6*bo*mu*lt)

def transient_ipr_gas(k, h, t_R, phi, mu, ct, rw, skin,
                       t_hours, q_mscf, m_pi, m_pwf) -> float:
    """Eq. 3.3 -- residual."""
    lt = (math.log10(t_hours) + math.log10(k/(phi*mu*ct*rw**2))
          - 3.23 + 0.87*skin)
    return q_mscf*1e3*1638*t_R*lt - k*h*(m_pi - m_pwf)

def joski_horizontal_ipr(kH, kV, h, bo, mu, L_ft, reH_ft, rw_ft,
                          pe, pwf, skin=0.0) -> float:
    """Eq. 3.11."""
    Iani = math.sqrt(kH/kV)
    a = (L_ft/2.0)*math.sqrt(0.5 + math.sqrt(
        0.25 + (reH_ft/(L_ft/2.0))**4))
    L2 = L_ft/2.0
    l1 = math.log((a + math.sqrt(a*a - L2*L2))/L2)
    l2 = (Iani*h/L_ft)*math.log(Iani*h/(rw_ft*(Iani*h + 1.0)))
    return kH*h*(pe - pwf)/(141.2*bo*mu*(l1 + l2 + skin))

def composite_ipr_single_phase(layer_data) -> float:
    """Eq. 3.40."""
    return sum(L["J_star"]*L["p_bar"] for L in layer_data)

def composite_ipr_two_phase(layer_data) -> float:
    """Eq. 3.43."""
    return sum(L["J_star"]*L["p_bar"]/1.8 for L in layer_data)

def composite_ipr_two_phase_rate(layer_data, pwf) -> float:
    """Eq. 3.42."""
    tot = 0.0
    for L in layer_data:
        j, p, pb = L["J_star"], L["p_bar"], L["p_b"]
        if pwf >= pb:
            tot += j*(p - pwf)
        else:
            qb = j*(p - pb); qv = j*pb/1.8; r = pwf/pb
            tot += qb + qv*(1 - 0.2*r - 0.8*r*r)
    return tot

def composite_pwf_two_phase(layer_data, q_total, p_lo=0.0, p_hi=None):
    from scipy.optimize import brentq
    if p_hi is None:
        p_hi = max(L["p_bar"] for L in layer_data)*0.999
    def res(p):
        return composite_ipr_two_phase_rate(layer_data, p) - q_total
    return brentq(res, p_lo, p_hi, xtol=1e-6)

def layer_ji_from_test(qi, pi, pwfi, pb=None) -> float:
    if pb is None or pwfi >= pb:
        return qi/(pi - pwfi)
    return qi/((pi - pb) + (pb/1.8)*(
        1 - 0.2*(pwfi/pb) - 0.8*(pwfi/pb)**2))

def future_j_vogel(j_present, kro_f, mu_f, bo_f, kro_p, mu_p, bo_p):
    return j_present*(kro_f/(bo_f*mu_f))/(kro_p/(bo_p*mu_p))

def future_rate_vogel(j_future, p_future, pwf):
    r = pwf/p_future
    return (j_future*p_future/1.8)*(1 - 0.2*r - 0.8*r**2)

def future_j_fetkovich(j_initial, p_future, p_initial):
    return j_initial*p_future/p_initial

import math
import numpy as np

def friction_chen(re, rel_rough) -> float:
    """Eq. 4.5."""
    t1 = rel_rough/3.7065
    t2 = (5.0452/re)*math.log10(
        rel_rough**1.1098/2.8257 + (7.149/re)**0.8981)
    return 1.0/(-4*math.log10(t1 - t2))**2

def bhp_average_tz(phf, q_sc, gamma_g, d_i, L, theta_deg,
                    t_avg_R, z_avg, fM) -> float:
    """Eq. 4.54."""
    s = 0.0375*gamma_g*L*math.cos(math.radians(theta_deg))/(z_avg*t_avg_R)
    es = math.exp(s)
    return math.sqrt(es*phf**2 +
        (6.67e-4*(es - 1)*fM*q_sc**2*z_avg**2*t_avg_R**2)
        /(d_i**5*math.cos(math.radians(theta_deg))))

def cullender_smith_bhp(phf, t_head_F, t_bh_F, d_in, L_ft, q_msc,
                          gamma_g, rel_rough=6e-4, theta_deg=0.0):
    """Eqs. 4.59-4.68."""
    from scipy.optimize import brentq
    fM = (1.0/(1.74 - 2.0*math.log10(2.0*rel_rough/d_in)))**2
    def I(p, t_F):
        tR = t_F + 460.0
        return ((p/tR)/(0.001*math.cos(math.radians(theta_deg))
                *(p/tR)**2 + 0.6666*fM*q_msc**2/d_in**5))
    half = 18.75*gamma_g*L_ft/2.0
    def rm(pmid):
        tmid = 0.5*(t_head_F + t_bh_F)
        return half - (pmid - phf)*(I(pmid, tmid) + I(phf, t_head_F))
    pmid = brentq(rm, phf, phf*5 + 1000, xtol=1e-3)
    def rb(pbh):
        return half - (pbh - pmid)*(I(pbh, t_bh_F)
                                    + I(pmid, 0.5*(t_head_F+t_bh_F)))
    return brentq(rb, pmid, pmid*5 + 1000, xtol=1e-3)

def poettmann_carpenter_bhp(phf, t_head_F, t_bh_F, d_in, depth_ft,
                              q_liq, wc, glr, api, gamma_w, gamma_g,
                              dz=100.0):
    """Eqs. 4.8-4.17."""
    go = 141.5/(api + 131.5)
    p = phf; h = 0.0
    while h < depth_ft:
        step = min(dz, depth_ft - h)
        tF = t_head_F + (t_bh_F - t_head_F)*(h + step/2)/depth_ft
        tR = tF + 460.0
        rs = gamma_g*((p/18)*10**(0.0125*api)/10**(0.00091*tF))**1.2048
        bo = 0.9759 + 0.00012*(rs*(gamma_g/go)**0.5 + 1.25*tF)**1.2
        M = 350.17*(go + wc*gamma_w) + glr*0.0764*gamma_g
        vm = 5.615*(bo + wc) + (glr - rs)*(14.7/p)*(tR/520.0)
        rho = M/vm
        dpv = 1.4737e-5*M*q_liq/(d_in/12)
        f2 = 10**(1.444 - 2.5*math.log10(dpv))
        kh = f2*q_liq**2*M**2/(7.4137e10*(d_in/12)**5)
        p += (rho + kh/rho)*step/144.0
        h += step
    return p

def guo_ghalambor_bhp_robust(phf, t_head_F, t_bh_F, d_in, L_ft,
                              theta_deg, q_g, q_o, q_w, q_s,
                              gamma_g, gamma_o, gamma_w, gamma_s,
                              rel_rough=6e-4):
    """Eq. 4.18."""
    from scipy.optimize import brentq
    tR = 0.5*(t_head_F + t_bh_F) + 460.0
    A = math.pi*(d_in/2/12)**2
    dh = d_in/12
    fM = (1.0/(1.74 - 2.0*math.log10(2.0*rel_rough/d_in)))**2
    a = (15.33*gamma_s*q_s + 86.07*gamma_w*q_w
         + 86.07*gamma_o*q_o + 18.79*gamma_g*q_g)/(1e3*tR*q_g)
    b = (0.2456*q_s + 1.379*q_w + 1.379*q_o)/(1e3*tR*q_g)
    c = 6.785e-6*tR*q_g/A
    d = (q_s + 5.615*(q_w + q_o))/(600.0*A)
    e = 6.0*fM/(2*32.17*dh)
    ct = math.cos(math.radians(theta_deg))
    den = ct + d*d*e
    M = c*d*e/den; N = c*c*e*ct/den**2; sn = math.sqrt(N)
    def res(p):
        arg = (144*p + M)**2 + N
        arg_hf = (144*phf + M)**2 + N
        t1 = 144*b*(p - phf)
        t2 = 0.5*(1 - 2*b*M)*math.log(arg/arg_hf)
        coef = (M + b*N/c - b*M**2)/sn
        t3 = coef*(math.atan((144*p+M)/sn) - math.atan((144*phf+M)/sn))
        return t1 + t2 - t3 - a*(1 + d*d*e)*L_ft
    return brentq(res, phf, phf*20 + 10000, xtol=1e-4)

def _mhb_cnl(N_L):
    X1 = math.log10(N_L + 3.0)
    Y = (-2.69851 + 0.15841*X1 - 0.55100*X1**2
         + 0.54785*X1**3 - 0.12195*X1**4)
    return 10**Y

def _mhb_yL_psi(X2):
    x = math.log10(X2) + 6.0
    return (-0.10307 + 0.61777*x - 0.63295*x**2
            + 0.29598*x**3 - 0.0401*x**4)

def _mhb_psi(X3):
    if X3 <= 0.01: return 1.0
    return (0.91163 - 4.82176*X3 + 1232.25*X3**2
            - 22253.6*X3**3 + 116174.3*X3**4)

def mhb_liquid_holdup(N_VL, N_VG, N_D, N_L, p, rho_L=62.4):
    cnl = _mhb_cnl(N_L)
    x2 = N_VL*p**0.1*cnl/(N_VG**0.575*14.7**0.1*N_D)
    ypsi = _mhb_yL_psi(x2)
    x3 = N_VG*N_L**0.38/N_D**2.14
    return ypsi*_mhb_psi(x3)

def mhb_pressure_traverse_full(phf, t_head_F, t_bh_F, d_in, L_ft,
                                 q_liq, glr, wc, api, gamma_g,
                                 gamma_w=1.05, sigma=30.0,
                                 mu_l=5.0, dz=50.0) -> np.ndarray:
    """Full mH-B traverse."""
    go = 141.5/(api + 131.5)
    dh = d_in/12; A = math.pi*(dh/2)**2
    rho_L = 62.4*go
    pts = [(0.0, phf)]
    p = phf; h = 0.0
    while h < L_ft:
        step = min(dz, L_ft - h)
        tF = t_head_F + (t_bh_F - t_head_F)*(h + step/2)/L_ft
        tR = tF + 460.0
        rs = gamma_g*((p/18)*10**(0.0125*api)/10**(0.00091*tF))**1.2048
        bo = 0.9759 + 0.00012*(rs*(gamma_g/go)**0.5 + 1.25*tF)**1.2
        rho_G = 2.7*gamma_g*p/(1.0*tR)
        usl = q_liq*(bo + wc)/(86400*A)
        qgres = q_liq*glr*(tR/520)*(14.7/p)
        usg = qgres/(86400*A)
        NVL = 1.938*usl*math.sqrt(rho_L/sigma)
        NVG = 1.938*usg*math.sqrt(rho_L/sigma)
        ND = 120.872*dh*math.sqrt(rho_L/sigma)
        NL = 0.15726*mu_l*math.sqrt(1.0/(rho_L*sigma**3))
        yL = mhb_liquid_holdup(NVL, NVG, ND, NL, p, rho_L)
        um = usl + usg
        yLns = usl/um if um > 0 else 1.0
        yL = max(yL, yLns)
        rho_m = yL*rho_L + (1-yL)*rho_G
        lamG = usg/um if um > 0 else 0.0
        LB = max(1.071 - 0.2218*um*um/dh, 0.13)
        if lamG < LB:
            u_s = 0.8
            inner = (1 + um/u_s)**2 - 4*usg/u_s
            yL = 1.0 - 0.5*(1 + um/u_s - math.sqrt(max(inner, 0)))
            ml = rho_L*q_liq*bo/86400.0
            Nre = max(2.2e-2*ml/(dh*mu_l), 2100)
        else:
            mu_mix = (mu_l**yL)*(0.02**(1-yL))
            Nre = max(2.2e-2*(q_liq*(bo+wc)*rho_m/86400)
                      /(dh*mu_mix), 2100)
        fF = friction_chen(Nre, 6e-4)
        dpf = 2*fF*rho_m*um*um/(32.17*dh)
        p += (rho_m + dpf)/144.0*step
        h += step
        pts.append((h, p))
    return np.array(pts)

def mist_flow_bhp(phf, t_head_F, t_bh_F, d_in, L_ft, theta_deg,

                          rho_g=2.0, sigma=30.0) -> str:
    vm = u_sl + u_sg
    if vm < 1e-9: return "bubble"
    Nvg = 1.938*u_sg*math.sqrt(rho_l/sigma)
    if u_sg < 0.5*u_sl + 0.1: return "bubble"
    if Nvg > 50 and u_sl < 5: return "annular"

def critical_ratio(k) -> float:
    """Eq. 5.1."""
    return (2.0/(k+1.0))**(k/(k-1.0))

def q_sonic_gas(cd, A2_in2, p_up, t_up_R, gamma_g, k) -> float:
    """Eq. 5.8. [Mscf/day]"""
    return 879*cd*A2_in2*p_up*math.sqrt(
        (k/(gamma_g*t_up_R))*(2.0/(k+1.0))**((k+1.0)/(k-1.0)))

def q_subsonic_gas(cd, A2_in2, p_up, p_dn, t_up_R, gamma_g, k) -> float:
    """Eq. 5.5. [Mscf/day]"""
    r = p_dn/p_up
    term = r**(2.0/k) - r**((k+1.0)/k)
    return 1248*cd*A2_in2*p_up*math.sqrt(
        (k/((k-1.0)*gamma_g*t_up_R))*term)

def q_liquid(cd, d2_in, dp_psi, rho) -> float:

    return 8074*cd*d2_in**2*math.sqrt(dp_psi/rho)

    """Eq. 5.12. [psia]"""

                                   z_up=1.0, z_out=1.0) -> float:

def sachdeva_critical_ratio(inp, tol=1e-9, max_iter=100) -> float:
    """Eq. 5.13."""
    from scipy.optimize import brentq
    k = inp.Cp_gas/inp.Cv_gas
    n = 1 + inp.x1*(inp.Cp_gas - inp.Cv_gas)/(
        inp.x1*inp.Cv_gas + (1-inp.x1)*inp.Cp_liquid)
    rhoL = inp.gamma_L*62.4; VL = 1.0/rhoL
    MWg = 28.97*inp.gamma_g
    VG1 = 10.73*(inp.t1_F + 460)/(inp.p1*MWg)
    def VG2(y): return VG1*y**(-1.0/k)
    def res(y):
        num = (k/(k-1)) + (1-inp.x1)*VL*(1-y)/(inp.x1*VG1)
        den = ((k-1) + n/2 + n*(1-inp.x1)*VL/(inp.x1*VG2(y))
               + (n/2)*((1-inp.x1)*VL/(inp.x1*VG2(y)))**2)
        return num/den - y**(k-1)
    return brentq(res, 1e-6, 1-1e-6, xtol=tol, maxiter=max_iter)

def sachdeva_flow(inp, max_iter=50, tol=1e-4) -> dict:
    """Full Sachdeva multiphase choke."""
    from .ch02_pvt import z_hall_yarborough, pseudo_critical_from_sg
    y_c = sachdeva_critical_ratio(inp)
    y_a = inp.p2/inp.p1
    y_u = y_c if y_a < y_c else y_a
    regime = "critical" if y_a < y_c else "subcritical"
    p_pc, t_pc = pseudo_critical_from_sg(inp.gamma_g)
    zv = 0.95; G2 = 0.0; q_liq = q_gas = 0.0
    for _ in range(max_iter):
        k = inp.Cp_gas/inp.Cv_gas
        rhoL = inp.gamma_L*62.4; VL = 1.0/rhoL
        MWg = 28.97*inp.gamma_g
        VG1 = zv*10.73*(inp.t1_F+460)/(inp.p1*MWg)
        VG2 = VG1*y_u**(-1.0/k)
        rho_m2 = 1.0/(inp.x1*VG2 + (1-inp.x1)*VL)
        bracket = ((1-inp.x1)*(1-y_u)/rhoL
                   + (inp.x1*k/(k-1))*(VG1 - y_u*VG2))
        G2 = inp.cd*math.sqrt(288*32.174*inp.p1*144*rho_m2*bracket)
        A2 = math.pi*(inp.d2_in/2/12)**2
        m = G2*A2
        mliq = (1-inp.x1)*m; mgas = inp.x1*m
        q_liq = mliq*86400/(rhoL*5.615)
        t2R = inp.t2_F + 460
        q_gas = mgas*10.73*t2R/(MWg*inp.p2)*86400/1e6
        zn = float(z_hall_yarborough(inp.p2/p_pc, t2R/t_pc))
        if abs(zn - zv) < tol: break
        zv = zn
    return dict(y_c=y_c, y_a=y_a, regime=regime, G2=G2,
                q_liq=q_liq, q_gas=q_gas, z_used=zv)

import math
import numpy as np

@dataclass
class ForecastStep:
    pressure: float
    rate: float
    cum_prod: float
    time_days: float

@dataclass
class ForecastResult:
    steps: list[ForecastStep] = field(default_factory=list)

    def to_table(self):

@dataclass
class PVTTable:
    pressures: np.ndarray
    Rs: np.ndarray
    Bo: np.ndarray
    Bg: np.ndarray
    mu_o: np.ndarray
    kro: np.ndarray
    krg: np.ndarray

    def interp(self, p):
        return {
            "Rs": float(np.interp(p, self.pressures, self.Rs)),
            "Bo": float(np.interp(p, self.pressures, self.Bo)),
            "Bg": float(np.interp(p, self.pressures, self.Bg)),
            "mu_o": float(np.interp(p, self.pressures, self.mu_o)),
            "kro": float(np.interp(p, self.pressures, self.kro)),
            "krg": float(np.interp(p, self.pressures, self.krg)),

def transient_forecast_oil(k, h, bo, mu, phi, ct, rw, skin,
                             p_i, t_months, tpr_func):
    from scipy.optimize import brentq
    from .ch03_ipr import transient_ipr_oil
    res = ForecastResult()
    cum = 0.0
    for m in range(1, t_months + 1):
        th = m*30*24.0
        def rq(q):
            return transient_ipr_oil(p_i, k, h, bo, mu, phi, ct,
                                      rw, skin, th, q) - tpr_func(q)
        try: q = brentq(rq, 0.1, 5000.0, xtol=1e-3)
        except ValueError: q = 0.0
        cum += q*30
        res.steps.append(ForecastStep(
            transient_ipr_oil(p_i, k, h, bo, mu, phi, ct,
                                rw, skin, th, q),
            q, cum, m*30))
    return res

def pseudo_steady_1phase_oil(p_initial, p_bubble, p_abandon,
                               j_star, bo, ct, ooip, tpr_func,
                               n_steps=10):
    from scipy.optimize import brentq
    pr = np.linspace(p_initial, max(p_abandon, p_bubble), n_steps)
    res = ForecastResult()
    cum = 0.0; t = 0.0
    for i in range(len(pr)-1):
        p0, p1 = pr[i], pr[i+1]
        pbar = 0.5*(p0 + p1)
        def rq(q): return j_star*(pbar - tpr_func(q)) - q
        try: q = brentq(rq, 1e-3, j_star*pbar*0.99, xtol=1e-4)
        except ValueError: q = 0.0
        r0 = np.exp(ct*(p_initial - p0)) - 1
        r1 = np.exp(ct*(p_initial - p1)) - 1
        dnp = (r1 - r0)*ooip
        dt = dnp/q if q > 0 else 0.0
        cum += dnp; t += dt
        res.steps.append(ForecastStep(pbar, q, cum, t))
    return res

def tarver_two_phase(p_initial, p_abandon, swi, N_ooip, pvt,
                      n_steps=10, max_iter=200, tol=1e-8):
    pr = np.linspace(p_initial, p_abandon, n_steps)
    Np = [0.0]; Gp = [0.0]; R_h = [float(pvt.interp(p_initial)["Rs"])]
    So_h = [1.0 - swi]
    for i in range(1, n_steps):
        p0, p1 = pr[i-1], pr[i]
        d0, d1 = pvt.interp(p0), pvt.interp(p1)
        Bo0, Bo1 = d0["Bo"], d1["Bo"]
        Bg0 = d0["Bg"]; Rs0, Rs1 = d0["Rs"], d1["Rs"]
        dB = Bo0 - Bo1; dR = Rs1 - Rs0
        den = dB + dR*Bg0
        phi_o = (Bo0 - Rs0*Bg0)/den if den else 0.0
        phi_g = Bg0/den if den else 0.0
        R = Rs1
        for _ in range(max_iter):
            Np_p = Np[-1]; Gp_p = Gp[-1]
            dN = (1 - phi_o*Np_p - phi_g*Gp_p)/(phi_o + R*phi_g)
            Np_n = Np_p + dN; Gp_n = Gp_p + dN*R
            So = (Bo1/Bo0)*(1-swi)*(1-Np_n)
            kro = d1["kro"]; krg = d1["krg"]
            Rn = Rs1 + (krg/kro)*(Bo1/Bg0 if Bg0 > 0 else 0)
            if abs(Rn - R)/max(R, 1.0) < tol: R = Rn; break
            R = Rn
        Np.append(Np_n); Gp.append(Gp_n); R_h.append(R); So_h.append(So)
    return {"pressures": pr, "Np": np.array(Np)*N_ooip,
            "Gp": np.array(Gp)*N_ooip,
            "R": np.array(R_h), "So": np.array(So_h)}

def gas_forecast_pseudo_steady(k, h, t_R, phi, ct, rw, re,

"""Chapter 8 -- Production Decline Analysis."""
import math
import numpy as np

def exponential_rate(qi, b, t): return qi*math.exp(-b*t)
def exponential_cum(qi, b, t): return (qi/b)*(1 - math.exp(-b*t))
def exponential_b_from_ln(q1, q2, t1, t2):
    return math.log(q1/q2)/(t2 - t1)
def exponential_b_from_np(q1, q2, np1, np2):
    return (q1 - q2)/(np2 - np1)
def harmonic_rate(q0, b, t): return q0/(1 + b*t)
def harmonic_cum(q0, b, t): return (q0/b)*math.log(1 + b*t)
def hyperbolic_rate(q0, a, b, t): return q0/(1 + (b/a)*t)**a
def hyperbolic_cum(q0, a, b, t):
    return (a*q0/(b*(a-1)))*(1 - (1 + (b/a)*t)**(1-a))

def identify_model(t, q) -> str:
    q = np.asarray(q, dtype=float)
    t = np.asarray(t, dtype=float)
    mask = (q > 0) & (t > 0)
    t, q = t[mask], q[mask]
    if len(t) < 5: return "unknown"
    def r2(yf, yt):
        sr = np.sum((yt - yf)**2)
        st = np.sum((yt - yt.mean())**2)
        return 1 - sr/st if st > 0 else 0
    lq = np.log(q); pe = np.polyfit(t, lq, 1)
    r2e = r2(np.polyval(pe, t), lq)
    rel = -np.gradient(q, t)/q
    if np.std(rel)/max(np.mean(rel), 1e-9) < 0.15:
        return "exponential"
    if r2e > 0.98: return "exponential"
    return "hyperbolic"

def diagnostic_derivative(t, dp):
    """Eq. 15.7."""
    t = np.asarray(t); dp = np.asarray(dp)

    pp = np.asarray(p_prime, dtype=float)

def ki_standing(p, tR, pc, Tb, Tc):
    a = 1.2 + 4.5e-4*p + 1.5e-9*p**2
    c = 0.89 - 1.7e-4*p - 3.5e-8*p**2
    bi = math.log10(pc/14.7)/(1.0/Tb - 1.0/Tc)
    Fi = bi*(1.0/Tb - 1.0/tR)

def flash_standing(z, p, tR, pc, Tb, Tc, MW):
    ki = np.array([ki_standing(p, tR, pc[i], Tb[i], Tc[i])
                   for i in range(len(z))])
    def rr(nv):
        return np.sum(z*(ki - 1)/(nv*(ki - 1) + 1))
    try: nv = brentq(rr, 1e-9, 1 - 1e-9, xtol=1e-12)
    except ValueError: nv = 1.0 if rr(0.5) > 0 else 0.0
    nl = 1.0 - nv
    x = z/(nl + ki*nv); y = ki*x
    return {"nv": nv, "nl": nl, "x": x, "y": y,
            "MW_L": float(np.sum(x*MW)),
            "MW_V": float(np.sum(y*MW)), "ki": ki}

def series_capacity_ratio(L, D1, L1, D2):
    return math.sqrt((L/D1**(16/3))
                      /(L1/D1**(16/3) + (L - L1)/D2**(16/3)))

def parallel_capacity_ratio(D1, D2):
    return ((math.sqrt(D1**(16/3)) + math.sqrt(D2**(16/3)))

    den = (L1/(math.sqrt(D1**(16/3)) + math.sqrt(D2**(16/3)))**2

def npv(cfs, i):

def optimize_lift_gas_allocation(wells, total_gas,
                                  max_per_well=float("inf")):
    n = len(wells)
    def neg(x): return -sum(wells[i]["q_vs_gas"](x[i])
                            for i in range(n))
    def cons(x): return total_gas - sum(x)
    bounds = [(0, min(max_per_well, total_gas))]*n
    x0 = np.full(n, total_gas/n)
    res = minimize(neg, x0, method="SLSQP", bounds=bounds,
                   constraints=[{"type": "ineq", "fun": cons}],
                   options={"ftol": 1e-6, "maxiter": 200})
    return {"allocation": res.x.tolist(),
            "total_oil": float(-res.fun), "success": res.success}

"""Sensitivity and Monte Carlo."""
from dataclasses import dataclass
from typing import Callable
import numpy as np
import pandas as pd

@dataclass
class Parameter:
    name: str
    nominal: float
    low: float
    high: float
    distribution: str = "uniform"

def one_at_a_time(func, base_params, params, n_points=10):
    rows = []
    for p in params:
        for v in np.linspace(p.low, p.high, n_points):
            a = dict(base_params); a[p.name] = v
            try: r = func(**a)
            except Exception: r = np.nan
            rows.append({"parameter": p.name, "value": v, "result": r})
    return pd.DataFrame(rows)

def monte_carlo(func, params, n_samples=1000, seed=42):
    rng = np.random.default_rng(seed)
    samples = {}
    for p in params:
        if p.distribution == "uniform":
            samples[p.name] = rng.uniform(p.low, p.high, n_samples)
        elif p.distribution == "triangular":
            samples[p.name] = rng.triangular(p.low, p.nominal,
                                             p.high, n_samples)
        elif p.distribution == "normal":
            sigma = (p.high - p.low)/6.0
            samples[p.name] = rng.normal(p.nominal, sigma, n_samples)
    results = np.array([func(**{k: samples[k][i] for k in samples})
                        for i in range(n_samples)])
    return {"results": results,
            "p10": float(np.percentile(results, 10)),
            "p50": float(np.percentile(results, 50)),
            "p90": float(np.percentile(results, 90)),
            "mean": float(results.mean()),
            "std": float(results.std())}

"""PVT import from CSV/XLSX."""
from pathlib import Path
import pandas as pd
import numpy as np
from ..ch07_forecast import PVTTable

REQUIRED = {"pressure_psia", "rs_scf_stb", "bo_rb_stb"}

def load_pvt(path):
    p = Path(path)
    if p.suffix.lower() in (".xlsx", ".xls"):
        df = pd.read_excel(p)
    elif p.suffix.lower() == ".csv":
        df = pd.read_csv(p)
    else:
        raise ValueError(f"Unsupported format: {p.suffix}")
    missing = REQUIRED - set(df.columns)
    if missing:
        raise ValueError(f"Missing columns: {missing}")
    n = len(df)
    return PVTTable(
        pressures=df["pressure_psia"].values,
        Rs=df["rs_scf_stb"].values,
        Bo=df["bo_rb_stb"].values,
        Bg=df.get("bg_rb_scf", pd.Series([0.0]*n)).values,
        mu_o=df.get("mu_cp", pd.Series([1.0]*n)).values,
        kro=df.get("kro", pd.Series([1.0]*n)).values,
        krg=df.get("krg", pd.Series([0.0]*n)).values,
    )

def validate_pvt(t):
    w = []
    if np.any(np.diff(t.pressures) < 0):
        w.append("Pressures must be monotonic")
    if np.any(t.Bo <= 0):
        w.append("B_o must be positive")
    if np.any(t.Rs < 0):
        w.append("R_s must be non-negative")
    return w
